#### 4.1 Platform SDK расширение
```javascript
// docs/_platform/platform.js

class BatmGamesPlatform {
  constructor() {
    // ...existing code...
    this.audioContext = null;
    this.musicNode = null;
    this.sounds = {};
    this.musicVolume = 0.5;
    this.sfxVolume = 0.7;
  }

  initAudio() {
    this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
    this.loadSounds();
  }

  async loadSounds() {
    const soundFiles = {
      'jump': '/sounds/jump.mp3',
      'coin': '/sounds/coin.mp3',
      'gameover': '/sounds/gameover.mp3',
      'click': '/sounds/click.mp3',
    };

    for (const [name, url] of Object.entries(soundFiles)) {
      const response = await fetch(url);
      const arrayBuffer = await response.arrayBuffer();
      this.sounds[name] = await this.audioContext.decodeAudioData(arrayBuffer);
    }
  }

  playSFX(name, volume = 1.0) {
    if (!this.sounds[name]) return;
    
    const source = this.audioContext.createBufferSource();
    const gainNode = this.audioContext.createGain();
    
    source.buffer = this.sounds[name];
    gainNode.gain.value = volume * this.sfxVolume;
    
    source.connect(gainNode);
    gainNode.connect(this.audioContext.destination);
    source.start(0);
  }

  async playMusic(trackName, loop = true) {
    // Implement music playback
  }

  stopMusic() {
    if (this.musicNode) {
      this.musicNode.stop();
      this.musicNode = null;
    }
  }
}
```

## 🚨 Известные риски

### Риск 1: GitHub Pages пути могут сломаться
**Вероятность:** СРЕДНЯЯ  
**Митигация:** Всегда использовать относительные пути, тестировать после каждого изменения структуры

### Риск 2: Canvas performance на старых телефонах
**Вероятность:** ВЫСОКАЯ  
**Митигация:** 
- FPS throttling (30 FPS на weak devices)
- Упрощённый рендер для mobile
- Feature detection: `if (isMobile && isLowEnd) { simplifiedMode = true; }`

### Риск 3: Audio не работает без user interaction
**Вероятность:** ВЫСОКАЯ  
**Митигация:** 
- Первый звук играть только после клика/тапа
- Показывать иконку "звук выключен" пока нет interaction
- Resume AudioContext on first touch

### Риск 4: Конфликты в Git при работе втроём
**Вероятность:** СРЕДНЯЯ  
**Митигация:** 
- Строгий Git Flow (feature branches)
- Code review обязателен
- Чёткое разделение зон ответственности
- Merge conflicts решать через VS Code UI

### Риск 5: Deploy сломает production (инцидент 12.09)
**Вероятность:** НИЗКАЯ (но критичная)  
**Митигация:** 
- Staging environment перед production
- Проверка по телу, не по HTTP коду
- Rollback plan (backup nginx config)
- Мониторинг после deploy (curl каждые 5 мин)
